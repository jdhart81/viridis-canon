/-! ## 1. Basic definitions -/

section Defs

variable {ι : Type*} [Fintype ι] [DecidableEq ι]

/-- Real-valued indicator of a boolean. -/
def ind (b : Bool) : ℝ := if b then 1 else 0

@[simp] lemma ind_true : ind true = 1 := rfl

@[simp] lemma ind_false : ind false = 0 := rfl

lemma ind_nonneg (b : Bool) : 0 ≤ ind b := by cases b <;> simp

/-- `plug e σ b` is the state that assigns `b` to the edge `e` and follows `σ` elsewhere. -/
def plug (e : ι) (σ : {f : ι // f ≠ e} → Bool) (b : Bool) : ι → Bool :=
  fun f => if h : f = e then b else σ ⟨f, h⟩

omit [Fintype ι] in
@[simp] lemma plug_self (e : ι) (σ : {f : ι // f ≠ e} → Bool) (b : Bool) :
    plug e σ b e = b := by simp [plug]

omit [Fintype ι] in
lemma plug_of_ne {e f : ι} (h : f ≠ e) (σ : {f : ι // f ≠ e} → Bool) (b : Bool) :
    plug e σ b f = σ ⟨f, h⟩ := by simp [plug, h]

/-- The independent-edge (product) probability weight of a state. -/
def weight (p : ι → ℝ) (ω : ι → Bool) : ℝ := ∏ f, (if ω f then p f else 1 - p f)

/-- The product weight of the *other* edges, i.e. of a state of `ι \ {e}`. -/
def coWeight (e : ι) (p : ι → ℝ) (σ : {f : ι // f ≠ e} → Bool) : ℝ :=
  ∏ f : {f : ι // f ≠ e}, (if σ f then p (f : ι) else 1 - p (f : ι))

/-- System reliability under independent edge states with survival probabilities `p`.
For the two-terminal structure function of `TwoTerminal` below this is exactly the
two-terminal reliability of the network. -/
def reliability (φ : (ι → Bool) → Bool) (p : ι → ℝ) : ℝ := ∑ ω, weight p ω * ind (φ ω)

/-- The pivotal probability of edge `e`: the probability that the remaining edges are in a
state for which `e` is critical (the system works iff `e` works). -/
def pivotalProb (φ : (ι → Bool) → Bool) (p : ι → ℝ) (e : ι) : ℝ :=
  ∑ σ : {f : ι // f ≠ e} → Bool,
    coWeight e p σ * ind (φ (plug e σ true) && !(φ (plug e σ false)))

end Defs

lemma sum_split (e : ι) {M : Type*} [AddCommMonoid M] (F : (ι → Bool) → M) :
    ∑ ω, F ω = ∑ σ : {f : ι // f ≠ e} → Bool, ∑ b : Bool, F (plug e σ b) := by
  rw [← Equiv.sum_comp (Equiv.piSplitAt e (fun _ => Bool)).symm F, Fintype.sum_prod_type,
    Finset.sum_comm]
  refine Finset.sum_congr rfl (fun σ _ => Finset.sum_congr rfl (fun b _ => ?_))
  congr 1
  funext f
  simp only [Equiv.piSplitAt, plug, Equiv.coe_fn_symm_mk]
  split <;> simp_all

lemma prod_split (e : ι) (g : ι → ℝ) : ∏ f, g f = g e * ∏ f : {f : ι // f ≠ e}, g (f : ι) := by
  rw [Fintype.prod_eq_mul_prod_compl e g]
  congr 1
  exact Finset.prod_subtype _ (by simp) g

lemma weight_plug (e : ι) (p : ι → ℝ) (σ : {f : ι // f ≠ e} → Bool) (b : Bool) :
    weight p (plug e σ b) = (if b then p e else 1 - p e) * coWeight e p σ := by
  rw [weight, prod_split e]
  simp only [plug_self, coWeight]
  congr 1
  exact Finset.prod_congr rfl fun f _ => by rw [plug_of_ne f.2]

lemma coWeight_update (e : ι) (p : ι → ℝ) (t : ℝ) (σ : {f : ι // f ≠ e} → Bool) :
    coWeight e (Function.update p e t) σ = coWeight e p σ :=

lemma coWeight_nonneg {e : ι} {p : ι → ℝ} (hp : ∀ f, 0 ≤ p f ∧ p f ≤ 1)

  rcases hp (f : ι) with ⟨h0, h1⟩
  by_cases h : σ f <;> simp [h] <;> linarith

/-! ## 3. C1 — Theorem 1: affineness and the pivotal derivative -/

section C1

variable {ι : Type*} [Fintype ι] [DecidableEq ι]

omit [Fintype ι] in
/-- For a coherent (monotone) structure function, the one-edge difference of indicators is
the pivotality indicator. -/
lemma ind_sub_ind_eq_pivotal {φ : (ι → Bool) → Bool} (hφ : Monotone φ) (e : ι)
    (σ : {f : ι // f ≠ e} → Bool) :
    ind (φ (plug e σ true)) - ind (φ (plug e σ false))
      = ind (φ (plug e σ true) && !(φ (plug e σ false))) := by
  have hle : φ (plug e σ false) ≤ φ (plug e σ true) := by
    refine hφ ?_
    intro f
    by_cases h : f = e
    · subst h; simp
    · simp [plug_of_ne h]
  cases hT : φ (plug e σ true) <;> cases hF : φ (plug e σ false) <;> simp_all
  exact absurd hle (by decide)

/-- **C1 (Theorem 1).** For independent edge states and a coherent (monotone) structure
function, the reliability is an affine function of each single edge survival probability,
and its partial derivative with respect to that probability equals the pivotal probability
of that edge. -/
theorem reliability_derivative_eq_pivotal_probability
    (φ : (ι → Bool) → Bool) (hφ : Monotone φ) (p : ι → ℝ) (e : ι) :
    (∀ t : ℝ, reliability φ (Function.update p e t)
        = reliability φ (Function.update p e 0) + t * pivotalProb φ p e)
      ∧ HasDerivAt (fun t : ℝ => reliability φ (Function.update p e t))
          (pivotalProb φ p e) (p e) := by
  sorry

/-- The pivotal probability is exactly the difference of the reliabilities obtained by
forcing the edge up and forcing it down. -/
lemma pivotalProb_eq_reliability_diff
    (φ : (ι → Bool) → Bool) (hφ : Monotone φ) (p : ι → ℝ) (e : ι) :
    pivotalProb φ p e
      = reliability φ (Function.update p e 1) - reliability φ (Function.update p e 0) := by
  have h := (reliability_derivative_eq_pivotal_probability φ hφ p e).1 1
  linarith

lemma pivotalProb_nonneg (φ : (ι → Bool) → Bool) {p : ι → ℝ}
    (hp : ∀ f, 0 ≤ p f ∧ p f ≤ 1) (e : ι) : 0 ≤ pivotalProb φ p e :=
  Finset.sum_nonneg fun σ _ =>
    mul_nonneg (coWeight_nonneg hp σ) (ind_nonneg _)

end C1

/-! ## 5. C3 — Corollary 2: the infinitesimal linear-budget gain -/

section C3

variable {ι : Type*} [Fintype ι] [DecidableEq ι]

/-- The first-order gain of an infinitesimal upgrade direction `d`. -/
def firstOrderGain (φ : (ι → Bool) → Bool) (p : ι → ℝ) (d : ι → ℝ) : ℝ :=
  ∑ f, pivotalProb φ p f * d f

/-- The set of first-order gains achievable by nonnegative upgrade directions obeying the
linear budget `∑ c f * d f ≤ B`. -/
def feasibleGains (φ : (ι → Bool) → Bool) (p : ι → ℝ) (c : ι → ℝ) (B : ℝ) : Set ℝ :=
  {g | ∃ d : ι → ℝ, (∀ f, 0 ≤ d f) ∧ ∑ f, c f * d f ≤ B ∧ g = firstOrderGain φ p d}

/-- The best pivotality-to-cost ratio over all edges. -/

/-! ## 6. C4 — Proposition 1: signs of the mixed interaction for series and parallel pairs -/

section C4

variable {ι : Type*} [Fintype ι] [DecidableEq ι]

/-- The mixed second difference of the reliability in the two edges `e₁`, `e₂`. -/
def mixedDifference (φ : (ι → Bool) → Bool) (p : ι → ℝ) (e₁ e₂ : ι) : ℝ :=
  reliability φ (Function.update (Function.update p e₁ 1) e₂ 1)
    - reliability φ (Function.update (Function.update p e₁ 1) e₂ 0)
    - reliability φ (Function.update (Function.update p e₁ 0) e₂ 1)
    + reliability φ (Function.update (Function.update p e₁ 0) e₂ 0)

lemma reliability_update_boolean (φ : (ι → Bool) → Bool) (p : ι → ℝ) (e : ι) (b : Bool) :
    reliability φ (Function.update p e (if b then 1 else 0))
      = reliability (fun ω => φ (Function.update ω e b)) p := by
  rw [reliability_condition φ _ e, reliability_condition _ p e]
  refine Finset.sum_congr rfl fun σ _ => ?_
  rw [coWeight_update, Function.update_self]
  have hT : Function.update (plug e σ true) e b = plug e σ b := by
    funext f
    by_cases h : f = e
    · subst h; simp
    · simp [Function.update_of_ne h, plug_of_ne h]
  have hF : Function.update (plug e σ false) e b = plug e σ b := by
    funext f
    by_cases h : f = e
    · subst h; simp
    · simp [Function.update_of_ne h, plug_of_ne h]
  simp only [hT, hF]
  cases b <;> simp only [if_true, if_false, Bool.false_eq_true] <;> ring

/-- The reliability of a structure function which is positive on some state is positive,
provided every edge probability is strictly between `0` and `1`. -/

  have h2 : 0 < weight p ω₀ * ind (ψ ω₀) := by

  linarith

open Finset

lemma sum_fin2 (F : (Fin 2 → Bool) → ℝ) :
    ∑ ω, F ω = F ![false, false] + F ![false, true] + F ![true, false] + F ![true, true] := by
  have hc : ∀ a b : Bool, (Fin.cons a (Fin.cons b finZeroElim) : Fin 2 → Bool) = ![a, b] := by

  simp only [piFinTwoEquiv, Equiv.coe_fn_symm_mk, Fintype.sum_bool, hc]

/-- The two-edge series structure function: both edges are required. -/
def series2 : (Fin 2 → Bool) → Bool := fun ω => ω 0 && ω 1

/-- The two-edge parallel structure function: either edge suffices. -/
def parallel2 : (Fin 2 → Bool) → Bool := fun ω => ω 0 || ω 1

lemma series2_monotone : Monotone series2 := by
  intro ω ω' h
  have h0 := h 0
  have h1 := h 1
  simp only [series2]
  revert h0 h1
  cases ω 0 <;> cases ω 1 <;> cases ω' 0 <;> cases ω' 1 <;> simp

lemma parallel2_monotone : Monotone parallel2 := by
  intro ω ω' h
  have h0 := h 0
  have h1 := h 1
  simp only [parallel2]
  revert h0 h1
  cases ω 0 <;> cases ω 1 <;> cases ω' 0 <;> cases ω' 1 <;> simp

lemma reliability_fin2 (φ : (Fin 2 → Bool) → Bool) (q : Fin 2 → ℝ) :
    reliability φ q =
      (1 - q 0) * (1 - q 1) * ind (φ ![false, false])
        + (1 - q 0) * q 1 * ind (φ ![false, true])
        + q 0 * (1 - q 1) * ind (φ ![true, false])
        + q 0 * q 1 * ind (φ ![true, true]) := by
  rw [reliability, sum_fin2]
  simp only [weight, Fin.prod_univ_two, Matrix.cons_val_zero, Matrix.cons_val_one,
    if_true, if_false, Bool.false_eq_true]

/-! ## 8. C6 — Negative control: equal marginals do not identify reliability -/

section C6

open Finset

variable {ι : Type*} [Fintype ι] [DecidableEq ι]

/-- A (possibly correlated) joint distribution of edge states. -/
def IsProbDist (μ : (ι → Bool) → ℝ) : Prop := (∀ ω, 0 ≤ μ ω) ∧ ∑ ω, μ ω = 1

/-- The marginal survival probability of edge `e` under a joint distribution. -/
def edgeMarginal (μ : (ι → Bool) → ℝ) (e : ι) : ℝ := ∑ ω, μ ω * ind (ω e)

/-- Terminal reliability under a (possibly correlated) joint distribution. -/
def jointReliability (φ : (ι → Bool) → Bool) (μ : (ι → Bool) → ℝ) : ℝ :=
  ∑ ω, μ ω * ind (φ ω)

/-- The independent (product) model with survival probabilities `p`, viewed as a joint
distribution. -/
lemma jointReliability_weight (φ : (ι → Bool) → Bool) (p : ι → ℝ) :
    jointReliability φ (weight p) = reliability φ p := rfl

/-- The independent fair-coin distribution on two edges. -/
noncomputable def indepHalf : (Fin 2 → Bool) → ℝ := fun _ => 1 / 4

/-- The perfectly correlated distribution on two edges: both edges are up, or both are
down, with probability `1/2` each. -/
noncomputable def comonotoneHalf : (Fin 2 → Bool) → ℝ := fun ω => if ω 0 = ω 1 then 1 / 2 else 0

lemma indepHalf_eq_weight : indepHalf = weight (fun _ : Fin 2 => (1 : ℝ) / 2) := by
  funext ω
  simp only [indepHalf, weight, Fin.prod_univ_two]
  cases ω 0 <;> cases ω 1 <;> norm_num

/-- **C6 (negative control).** There are two joint distributions of edge states with
identical edge marginals but different terminal reliabilities for one and the same
coherent structure function; moreover one of the two is the independent product model
with those very marginals. Hence marginal edge survival probabilities do not identify
terminal reliability once failures may be correlated. -/
theorem equal_marginals_do_not_identify_reliability :
    ∃ (φ : (Fin 2 → Bool) → Bool) (p : Fin 2 → ℝ) (μ₁ μ₂ : (Fin 2 → Bool) → ℝ),
      Monotone φ ∧ IsProbDist μ₁ ∧ IsProbDist μ₂ ∧
      μ₁ = weight p ∧
      (∀ e, edgeMarginal μ₁ e = p e) ∧
      (∀ e, edgeMarginal μ₁ e = edgeMarginal μ₂ e) ∧
      jointReliability φ μ₁ ≠ jointReliability φ μ₂ := by
  sorry

end C6

conclusion. -/

\begin{abstract}
Ecological circuit models rank landscape elements by deterministic flow, while corridor failures introduce a different question: which marginal survival improvement most increases the probability that habitat terminals remain connected? For a finite undirected multigraph with independent Bernoulli edge states, we condition on one edge to show that the derivative of two-terminal reliability with respect to its survival probability equals its pivotal probability. Under an infinitesimal linear stewardship budget, pivotal probability per marginal cost is therefore the exact local priority. A three-edge adversarial example reverses the ranking from equal-conductance electrical current: the highest-current edge has pivotal probability $0.19$, while two lower-current edges each have pivotal probability $0.81$. Series and parallel controls yield opposite mixed-interaction signs, and equal-marginal correlated controls demonstrate that independence is indispensable. Complete enumeration over 54 small graphs, 500 seeded random graphs, and Monte Carlo convergence produced 13 passes and no failures. The derivative is established Birnbaum reliability importance; the HDFM specialization has unassessed novelty and requires empirical calibration and independent significance review.
\end{abstract}

\textbf{Keywords:} habitat corridors; reliability; pivotal probability; conservation planning; stewardship priority

\section{Introduction and preregistered prior work}

Connectivity models based on electrical circuits treat movement through a resistance landscape as current flow \cite{mcrae2008}. Current ecological work combines edge removal, patch stability, and network robustness to allocate limited conservation inputs \cite{jiang2024}; independent validation can materially change recommended corridors even when resistance surfaces are correlated \cite{poor2024}. Dynamic analyses further warn that static priorities can miss long-horizon climate and land-use exposure \cite{xu2026}. Budgeted generalized-flow models and robust green-bridge design address related finite optimization and single-failure protection questions \cite{hamonic2023,ellmies2026}.

This paper asks a narrower question under a declared independent-failure model. The relevant mathematical object is classical Birnbaum importance: for a coherent system, component critical probability equals a partial derivative of reliability \cite{borgonovo2010,aliee2017}. We do not claim a new reliability theorem. We test whether the specialization clarifies when deterministic corridor use and stochastic stewardship sensitivity can disagree.

Before conclusions, we preregistered six hypotheses, explicit falsifiers, exhaustive and randomized checks, topology controls, a ranking-reversal target, a dependence-breaking control, and a convergence test. The full record is distributed with the package.

\section{Model and exact local result}

Let $G=(V,E)$ be a finite undirected multigraph with terminals $s,t$. Each edge $e$ has independent operational indicator $X_e\sim\mathrm{Bernoulli}(q_e)$. Write $A$ for the increasing event that $s$ is connected to $t$, and
\[
R(\mathbf q)=\Pr_{\mathbf q}(A).
\]
For an edge $e$, let $R_e^+$ and $R_e^-$ denote reliability when $e$ is forced operational or failed while all other probabilities remain fixed. Edge $e$ is pivotal in an assignment of the other edges when forcing it up makes $A$ true and forcing it down makes $A$ false. Let $\pi_e$ be the probability of that event over the other edges.

\begin{theorem}[Pivotal corridor derivative]
For every edge $e$,
\[
R(\mathbf q)=q_eR_e^+ +(1-q_e)R_e^-,\qquad
\frac{\partial R}{\partial q_e}=R_e^+-R_e^-=\pi_e.
\]
\end{theorem}

\begin{proof}
Condition on $X_e$. The first equality follows by total probability. Differentiation gives $R_e^+-R_e^-$ because neither forced reliability depends on $q_e$. For each assignment of the other edges, monotonicity of connectivity makes the difference of the two connection indicators either zero or one; it is one exactly when $e$ is pivotal. Taking its expectation gives $\pi_e$.
\end{proof}

\begin{corollary}
$\pi_e$ is independent of $q_e$ itself, although it depends on all other edge probabilities.
\end{corollary}

Suppose an infinitesimal action $x_e\ge0$ raises survival probability at marginal cost $c_e>0$, with $\sum_ec_ex_e\le B$. The first-order reliability gain is $\sum_e\pi_ex_e$.

\begin{corollary}[Local stewardship rule]
\[
\max_{x\ge0,\,\sum c_ex_e\le B}\sum_e\pi_ex_e
=B\max_e\frac{\pi_e}{c_e}.
\]
Any edge attaining the largest ratio is a first-order optimum.
\end{corollary}

\begin{proof}
$\sum_e\pi_ex_e\le(\max_e\pi_e/c_e)\sum_ec_ex_e\le B\max_e\pi_e/c_e$. Allocate the infinitesimal budget to a maximizing edge for equality.
\end{proof}

This statement is deliberately local. Finite probability changes alter other edges' pivotalities, may hit bounds, and can require joint optimization.

\section{Topology and metric-separation controls}

\begin{proposition}[Opposite interaction signs]
For two required series edges, $R=q_1q_2$ and $\partial^2R/(\partial q_1\partial q_2)=+1$. For two parallel edges, $R=q_1+q_2-q_1q_2$ and the mixed derivative is $-1$.
\end{proposition}

Thus series stewardship effects are complementary while parallel effects are substitutable. A topology-free universal sign claim is false.

\begin{proposition}[Current--pivotality ranking reversal]
Consider terminals joined by a direct edge $e_0$ and a two-edge path $(e_1,e_2)$. Give every edge unit conductance and survival probabilities $(0.1,0.9,0.9)$. Unit electrical currents have magnitudes $(2/3,1/3,1/3)$, but pivotal probabilities are $(0.19,0.81,0.81)$.
\end{proposition}

\begin{proof}
Parallel branch conductances are $1$ and $1/2$, so one unit of flow splits $2/3$ versus $1/3$. The direct edge is pivotal only when the two-edge path fails, giving $1-0.9^2=0.19$. Each path edge is pivotal when the direct edge fails and its partner survives, giving $0.9\cdot0.9=0.81$.
\end{proof}

The example does not invalidate current-flow metrics. It separates two estimands: modeled physical use under resistance versus sensitivity of connection probability under failure risk. A field model may relate them, but equality cannot be assumed.

\section{Adversarial verification}

The standard-library Python gate enumerated every edge state. Forced-edge differences and direct pivotal-state sums agreed on all 54 connected simple terminal graphs on two through four vertices, with worst absolute discrepancy $2.22\times10^{-16}$. They also agreed on 500 seeded connected random graphs with three through six vertices and at most ten edges, with worst error $1.61\times10^{-15}$. Centered finite differences, own-probability invariance, series/parallel signs, 1,000 random linear-budget instances, and the ranking reversal passed.

For an eight-edge test graph, pivotality Monte Carlo RMSE across 24 seeded replicates fell from $8.28\times10^{-3}$ at 2,000 samples to $4.92\times10^{-3}$ at 8,000 and $1.91\times10^{-3}$ at 32,000. RMSE times $\sqrt N$ remained within a factor of 1.5. The final tally was 13 PASS and 0 FAIL. This numerical evidence is distinct from formal certification; the exact declared Lean statements are separately Comparator-certified.

\paragraph{Dependence-breaking negative control.}
Two series edges with identical marginals $q=0.9$ have reliability $0.81$ under independence but $0.9$ when their states are perfectly correlated. Two parallel edges have reliability $0.99$ under independence but $0.9$ under perfect correlation. Marginals alone therefore do not identify reliability, pivotality, or priority under shared hazards.

\section{Interpretation, limitations, and next tests}

The practical interpretation is conditional: if edge survival probabilities are calibrated and approximately independent, pivotality per marginal cost is the correct local sensitivity score for a declared terminal objective. It should be reported beside, not silently substituted for, movement-current, patch-stability, and finite-budget metrics. Shared wildfire, flood, climate, road, land-tenure, or governance exposure requires a joint hazard model. Static probabilities also cannot represent the temporal turnover documented in current connectivity research \cite{xu2026}.

Other limitations are substantial. Survival probabilities and costs are treated as known; only a Boolean two-terminal endpoint is modeled; habitat quality, capacity, dispersal, population dynamics, recovery, equity, and multi-species objectives are omitted; enumeration is exponential; the budget rule is infinitesimal; and no field data were used. A credible next test needs one real corridor graph, independent movement or persistence validation, explicit shared-hazard scenarios, and a finite-budget comparison against existing priorities \cite{poor2024}.

The private Comparator certified the exact frozen declarations covering affine edge dependence with the pivotal derivative, own-probability independence, the local budget bound, series/parallel mixed signs, and equal-marginal non-identification. Novelty and scientific significance remain unassessed.
